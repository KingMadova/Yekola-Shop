import { NextResponse } from "next/server";
import { isRecord, parseLeadPayload } from "@/lib/leads";

const WINDOW_MS = 10 * 60 * 1000;
const MAX_PER_WINDOW = 5;
const hits = new Map<string, number[]>();

/** Limite simple par IP (mémoire du serveur : suffisant contre les abus ordinaires). */
function isRateLimited(ip: string): boolean {
  const now = Date.now();
  const recent = (hits.get(ip) ?? []).filter((time) => now - time < WINDOW_MS);
  recent.push(now);
  hits.set(ip, recent);
  return recent.length > MAX_PER_WINDOW;
}

export async function POST(request: Request) {
  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return NextResponse.json({ ok: false }, { status: 400 });
  }

  // Champ piège invisible : seuls les robots le remplissent. On répond « ok » sans rien enregistrer.
  if (isRecord(body) && typeof body.website === "string" && body.website.trim() !== "") {
    return NextResponse.json({ ok: true });
  }

  const ip = request.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ?? "inconnue";
  if (isRateLimited(ip)) return NextResponse.json({ ok: false }, { status: 429 });

  const lead = parseLeadPayload(body);
  if (!lead) return NextResponse.json({ ok: false }, { status: 400 });

  const url = process.env.GOOGLE_SHEETS_WEBHOOK_URL;
  const secret = process.env.LEADS_WEBHOOK_SECRET;
  if (!url || !secret) {
    console.warn("[leads] GOOGLE_SHEETS_WEBHOOK_URL ou LEADS_WEBHOOK_SECRET manquant");
    return NextResponse.json({ ok: false }, { status: 503 });
  }

  try {
    const response = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify({ ...lead, token: secret }),
      redirect: "follow",
      cache: "no-store",
    });
    const result: unknown = await response.json().catch(() => null);
    if (!response.ok || !isRecord(result) || result.ok !== true) {
      throw new Error(`Réponse inattendue du script (${response.status})`);
    }
    return NextResponse.json({ ok: true });
  } catch (error) {
    console.error("[leads] échec de l'envoi vers Google Sheets", error);
    return NextResponse.json({ ok: false }, { status: 502 });
  }
}

import type { Metadata } from "next";
import Link from "next/link";
import { notFound } from "next/navigation";
import { ProductGrid } from "@/components/catalogue/ProductGrid";
import { ProductViewSignal } from "@/components/lead/ProductViewSignal";
import { ProductDetail } from "@/components/product/ProductDetail";
import { container } from "@/components/ui/styles";
import { PRODUCTS, getProductBySlug } from "@/data/products";
import { SITE_URL } from "@/lib/config";

interface ProductPageProps {
  params: Promise<{ slug: string }>;
}

export function generateStaticParams() {
  return PRODUCTS.map((product) => ({ slug: product.slug }));
}

export async function generateMetadata({ params }: ProductPageProps): Promise<Metadata> {
  const product = getProductBySlug((await params).slug);
  if (!product) return {};
  return {
    title: product.name,
    description: product.shortDesc,
    openGraph: { title: product.name, description: product.shortDesc, images: product.images.slice(0, 1) },
  };
}

export default async function ProductPage({ params }: ProductPageProps) {
  const { slug } = await params;
  const product = getProductBySlug(slug);
  if (!product) notFound();

  const related = PRODUCTS.filter(
    (other) =>
      other.code !== product.code &&
      (other.category === product.category || other.healthGoals.some((goal) => product.healthGoals.includes(goal))),
  ).slice(0, 4);

  const jsonLd = {
    "@context": "https://schema.org",
    "@type": "Product",
    name: product.name,
    description: product.shortDesc,
    sku: product.code,
    image: product.images.map((src) => new URL(src, SITE_URL).toString()),
    offers: {
      "@type": "Offer",
      priceCurrency: "XAF",
      price: product.pricePublic,
      availability: "https://schema.org/InStock",
      url: `${SITE_URL}/produit/${product.slug}`,
    },
  };

  return (
    <>

type TextField = "fullName" | "phone" | "neighborhood";
type FormErrors = Partial<Record<TextField, string>>;

const INITIAL_VALUES: CustomerInfo = {
  fullName: "",
  phone: "",
  zone: DELIVERY_ZONES[0],
  neighborhood: "",
  payment: PAYMENT_METHODS[1], // MTN Mobile Money, modifiable par le client
};

function validate(values: CustomerInfo): FormErrors {
  const errors: FormErrors = {};
  if (values.fullName.trim().length < 2) errors.fullName = "Indiquez votre prénom et votre nom.";
  if (!normalizeCongoPhone(values.phone)) {
    errors.phone = "Entrez un numéro WhatsApp valide, par exemple 06 123 4567.";
  }
  if (values.neighborhood.trim().length < 3) {
    errors.neighborhood = "Indiquez votre quartier ou un repère précis pour la livraison.";
  }
  return errors;
}

const controlClass =
  "min-h-12 w-full rounded-lg border bg-white px-3 text-base text-gray-900 focus-visible:outline-2 focus-visible:outline-offset-1 focus-visible:outline-emerald-700";

interface FieldProps extends InputHTMLAttributes<HTMLInputElement> {
  label: string;
  error?: string;
  inputRef: RefObject<HTMLInputElement | null>;
}

function Field({ label, error, inputRef, id, ...inputProps }: FieldProps) {
  const errorId = `${id}-error`;
  return (
    <div className="flex flex-col gap-1">
      <label htmlFor={id} className="text-sm font-semibold text-gray-900">
        {label}
      </label>
      <input
        {...inputProps}
        id={id}
        ref={inputRef}
        aria-invalid={error ? true : undefined}
        aria-describedby={error ? errorId : undefined}
        className={`${controlClass} ${error ? "border-red-700" : "border-gray-400"}`}
      />
      {error && (
        <p id={errorId} className="text-sm text-red-700">
          {error}
        </p>
      )}
    </div>
  );
}

interface CheckoutFormProps {
  items: readonly CartItem[];
  totals: CartTotals;
}

/** Formulaire de commande express : valide les champs, puis ouvre WhatsApp avec le récapitulatif. */
export function CheckoutForm({ items, totals }: CheckoutFormProps) {
  const uid = useId();
  const [values, setValues] = useState<CustomerInfo>(INITIAL_VALUES);
  const [errors, setErrors] = useState<FormErrors>({});
  const [sent, setSent] = useState(false);

  const fieldRefs: Record<TextField, RefObject<HTMLInputElement | null>> = {
    fullName: useRef<HTMLInputElement>(null),
    phone: useRef<HTMLInputElement>(null),
    neighborhood: useRef<HTMLInputElement>(null),
  };

  function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const nextErrors = validate(values);
    setErrors(nextErrors);

    const firstInvalid = (["fullName", "phone", "neighborhood"] as const).find(
      (field) => nextErrors[field],
    );
    if (firstInvalid) {
      fieldRefs[firstInvalid].current?.focus();
      return;
    }

    const url = buildWhatsAppUrl(buildCartOrderMessage(items, values));
    const popup = window.open(url, "_blank");
    if (popup) popup.opener = null;
    else window.location.assign(url); // pop-up bloqué : on ouvre WhatsApp dans l'onglet courant
    setSent(true);
  }

  return (
    <form onSubmit={handleSubmit} noValidate className="flex flex-col gap-4">
      <h3 className="text-lg font-bold text-gray-900">Vos informations de livraison</h3>

      <Field
        id={`${uid}-name`}
        label="Prénom et nom"
        autoComplete="name"
        value={values.fullName}
        onChange={(event) => setValues({ ...values, fullName: event.target.value })}
        error={errors.fullName}
        inputRef={fieldRefs.fullName}
      />
      <Field
        id={`${uid}-phone`}
        label="Téléphone WhatsApp"
        type="tel"
        inputMode="tel"
        autoComplete="tel"
        placeholder="06 123 4567"
        value={values.phone}
        onChange={(event) => setValues({ ...values, phone: event.target.value })}
        error={errors.phone}
        inputRef={fieldRefs.phone}
      />

      <div className="flex flex-col gap-1">
        <label htmlFor={`${uid}-zone`} className="text-sm font-semibold text-gray-900">
          Ville
        </label>
        <select
          id={`${uid}-zone`}
          value={values.zone}
          onChange={(event) =>
            setValues({ ...values, zone: event.target.value as CustomerInfo["zone"] })
          }
          className={`${controlClass} border-gray-400`}
        >
          {DELIVERY_ZONES.map((zone) => (
            <option key={zone} value={zone}>
              {zone}
            </option>
          ))}
        </select>
      </div>

      <Field
        id={`${uid}-area`}
        label="Quartier / repère précis"
        value={values.neighborhood}
        onChange={(event) => setValues({ ...values, neighborhood: event.target.value })}
        error={errors.neighborhood}
        inputRef={fieldRefs.neighborhood}
      />

      <fieldset className="flex flex-col gap-2">
        <legend className="mb-1 text-sm font-semibold text-gray-900">Mode de règlement</legend>
        {PAYMENT_METHODS.map((method) => (

          >
            <input
              type="radio"
              name={`${uid}-payment`}

              checked={values.payment === method}
              onChange={() => setValues({ ...values, payment: method })}
              className="size-4 accent-emerald-700"
            />
            {method}
          </label>
        ))}
      </fieldset>

"use client";

import { useEffect } from "react";
import { useInterestStore } from "@/store/useInterestStore";

/** Lit ?besoin=…&utm_… à l'arrivée du visiteur : le besoin est connu avant même qu'il ne clique. */
export function InterestTracker(): null {
  useEffect(() => {
    let referrer = "";
    try {
      const url = new URL(document.referrer);
      if (url.host !== window.location.host) referrer = url.hostname;
    } catch {
      /* pas de référent */
    }
    useInterestStore.getState().captureArrival({
      search: window.location.search,
      pathname: window.location.pathname,
      referrer,
    });
  }, []);
  return null;
}

const fieldClass = `min-h-12 w-full rounded-lg border bg-white px-3 text-base text-gray-900 ${focusRing}`;

function isTyping(): boolean {
  const element = document.activeElement;
  return (
    element instanceof HTMLInputElement ||
    element instanceof HTMLTextAreaElement ||
    element instanceof HTMLSelectElement
  );
}

const cartGoals = (): HealthGoal[] => getCartItems().flatMap((item) => item.product.healthGoals);

/**
 * Pop-up de capture : prénom + numéro WhatsApp, rien d'autre d'obligatoire.
 * Le besoin est déduit du parcours (publicité, filtres, fiches, panier) ; le visiteur peut le corriger en un geste.
 */
export function LeadCaptureModal() {
  const titleId = useId();
  const dialogRef = useRef<HTMLDialogElement>(null);
  const nameRef = useRef<HTMLInputElement>(null);
  const phoneRef = useRef<HTMLInputElement>(null);
  const submitted = useRef(false);

  const [open, setOpen] = useState(false);
  const [done, setDone] = useState(false);
  const [firstName, setFirstName] = useState("");
  const [phone, setPhone] = useState("");
  const [honeypot, setHoneypot] = useState("");
  const [need, setNeed] = useState<HealthGoal | null>(null);
  const [inferred, setInferred] = useState(false);
  const [editingNeed, setEditingNeed] = useState(false);
  const [touched, setTouched] = useState(false);
  const [errors, setErrors] = useState<{ firstName?: string; phone?: string }>({});

  // Ouverture après 7 s, sans jamais gêner : onglet masqué, panier ouvert ou saisie en cours => on attend.
  useEffect(() => {
    retryPendingLead();
    if (readLeadStatus() !== null) return;

    let timer = 0;
    let attempts = 0;
    const tryOpen = () => {
      const busy = document.hidden || useCartStore.getState().isOpen || isTyping();
      if (busy) {
        if (attempts < 8) {
          attempts += 1;
          timer = window.setTimeout(tryOpen, 4000);
        }
        return;
      }
      const interest = useInterestStore.getState();
      const profile = buildNeedProfile({
        scores: interest.scores,
        origins: interest.origins,
        chosen: null,
        cartGoals: cartGoals(),
      });
      setNeed(profile.need);
      setInferred(profile.need !== null);
      setOpen(true);
    };
    timer = window.setTimeout(tryOpen, LEAD_MODAL_DELAY_MS);
    return () => window.clearTimeout(timer);
  }, []);

  useEffect(() => {
    const dialog = dialogRef.current;
    if (open && dialog && !dialog.open) dialog.showModal();
  }, [open]);

  function handleClose() {
    setOpen(false);
    if (!submitted.current) writeLeadStatus("dismissed");
  }

  function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const normalized = normalizeCongoPhone(phone);
    const nextErrors = {
      firstName: firstName.trim().length < 2 ? "Indiquez votre prénom." : undefined,
      phone: normalized ? undefined : "Entrez un numéro WhatsApp valide, par exemple 06 123 4567.",
    };
    setErrors(nextErrors);
    if (nextErrors.firstName) return nameRef.current?.focus();
    if (nextErrors.phone || !normalized) return phoneRef.current?.focus();

    const interest = useInterestStore.getState();
    const items = getCartItems();
    const profile = buildNeedProfile({
      scores: interest.scores,
      origins: interest.origins,
      chosen: touched ? need : null,
      cartGoals: items.flatMap((item) => item.product.healthGoals),
    });
    const payload: LeadPayload = {
      firstName: firstName.trim(),
      phone: normalized,
      need: profile.need,
      needOrigin: profile.origins.map((origin) => ORIGIN_LABELS[origin]).join(", "),
      otherNeeds: profile.others,
      viewedProducts: interest.viewedProducts.map((code) => getProductByCode(code)?.name ?? code),
      cartTotal: computeCartTotals(items).total,
      campaign: interest.campaign ?? "",
      source: interest.source ?? "",
      referrer: interest.referrer ?? "",
      landingPath: interest.landingPath ?? "",
      pagePath: window.location.pathname,
    };

    // 1) Le lead part vers Google Sheets sans attendre (copie locale en cas d'échec). 2) WhatsApp s'ouvre aussitôt.
    sendLead(payload, honeypot);
    submitted.current = true;
    writeLeadStatus("submitted");

    const url = buildWhatsAppUrl(buildLeadMessage(payload.firstName, payload.phone, profile.need ?? undefined));
    const popup = window.open(url, "_blank");
    if (popup) popup.opener = null;
    else window.location.assign(url);
    setDone(true);
  }

  const chips = (
    <fieldset className="flex flex-col gap-2">
      <legend className="mb-1 text-sm font-semibold text-gray-900">Votre besoin (facultatif)</legend>
      <div className="flex flex-wrap gap-2">
        {HEALTH_GOALS.map((goal) => (
          <label
            key={goal}
            className="inline-flex min-h-11 cursor-pointer items-center rounded-full border border-gray-300 bg-white px-4 text-sm has-[:checked]:border-emerald-700 has-[:checked]:bg-emerald-700 has-[:checked]:text-white has-[:focus-visible]:outline-2 has-[:focus-visible]:outline-offset-2 has-[:focus-visible]:outline-emerald-700"
          >
            <input
              type="radio"
              name={`${titleId}-need`}
              value={goal}
              checked={need === goal}
              onChange={() => {
                setNeed(goal);
                setTouched(true);
              }}
              className="sr-only"
            />
            {HEALTH_GOAL_META[goal].shortLabel}
          </label>
        ))}
      </div>
    </fieldset>
  );

  return (
    <dialog
      ref={dialogRef}
      aria-labelledby={titleId}
      onClose={handleClose}
      onClick={(event) => {
        if (event.target === dialogRef.current) dialogRef.current?.close();
      }}
      className="m-0 mt-auto max-h-[92dvh] w-full max-w-none overflow-y-auto rounded-t-3xl bg-white p-0 text-gray-900 backdrop:bg-black/50 sm:m-auto sm:max-w-md sm:rounded-3xl"
    >
      <div className="relative flex flex-col gap-4 p-5">
        <button
          type="button"
          onClick={() => dialogRef.current?.close()}
          aria-label="Fermer"
          className={`absolute right-2 top-2 inline-flex size-11 items-center justify-center rounded-full hover:bg-gray-100 ${focusRing}`}
        >
          <X aria-hidden className="size-5" />
        </button>

        {done ? (
          <div role="status" className="flex flex-col items-center gap-3 py-4 text-center">
            <Check aria-hidden className="size-12 rounded-full bg-emerald-100 p-2.5 text-emerald-800" />
            <h2 id={titleId} className="text-xl font-bold">Merci {firstName.trim()} !</h2>
            <p className="text-gray-800">WhatsApp s&apos;ouvre avec votre demande. Envoyez le message pour recevoir le catalogue.</p>
            <button type="button" onClick={() => dialogRef.current?.close()} className={`min-h-12 rounded-xl border border-emerald-700 px-6 font-semibold text-emerald-800 hover:bg-emerald-50 ${focusRing}`}>
              Fermer
            </button>
          </div>
        ) : (
          <form onSubmit={handleSubmit} noValidate className="relative flex flex-col gap-4">
            <h2 id={titleId} className="pr-10 text-xl font-bold">
              Téléchargez le catalogue officiel Forever avec les tarifs en FCFA sur WhatsApp
            </h2>

            <div className="flex flex-col gap-1">
              <label htmlFor={`${titleId}-name`} className="text-sm font-semibold">Prénom</label>
              <input
                id={`${titleId}-name`}
                ref={nameRef}
                value={firstName}
                onChange={(event) => setFirstName(event.target.value)}
                autoComplete="given-name"
                aria-invalid={errors.firstName ? true : undefined}
                aria-describedby={errors.firstName ? `${titleId}-name-error` : undefined}
                className={`${fieldClass} ${errors.firstName ? "border-red-700" : "border-gray-400"}`}
              />
              {errors.firstName && <p id={`${titleId}-name-error`} className="text-sm text-red-700">{errors.firstName}</p>}
            </div>

            <div className="flex flex-col gap-1">
              <label htmlFor={`${titleId}-phone`} className="text-sm font-semibold">Numéro WhatsApp</label>
              <input
                id={`${titleId}-phone`}
                ref={phoneRef}
                type="tel"
                inputMode="tel"
                autoComplete="tel"
                placeholder="06 123 4567"
                value={phone}
                onChange={(event) => setPhone(event.target.value)}
                aria-invalid={errors.phone ? true : undefined}
                aria-describedby={errors.phone ? `${titleId}-phone-error` : undefined}
                className={`${fieldClass} ${errors.phone ? "border-red-700" : "border-gray-400"}`}
              />
              {errors.phone && <p id={`${titleId}-phone-error`} className="text-sm text-red-700">{errors.phone}</p>}
            </div>

            {inferred && need && !editingNeed ? (
              <p className="text-sm text-gray-800">
                Catalogue adapté à : <strong>{HEALTH_GOAL_META[need].shortLabel}</strong>{" "}
                <button type="button" onClick={() => setEditingNeed(true)} className={`rounded font-medium text-emerald-800 underline ${focusRing}`}>
                  Changer
                </button>
              </p>
            ) : (
              chips
            )}

            {/* Champ piège : invisible pour les humains, rempli par les robots. */}
            <input
              type="text"
              name="website"
              tabIndex={-1}
              autoComplete="off"
              aria-hidden="true"
              value={honeypot}
              onChange={(event) => setHoneypot(event.target.value)}
              className="absolute -left-[9999px] size-px opacity-0"
            />

            <button type="submit" className={`inline-flex min-h-12 items-center justify-center gap-2 rounded-xl bg-[#075E54] px-5 font-semibold text-white hover:bg-[#064c44] ${focusRing}`}>
              <MessageCircle aria-hidden className="size-5" />
              Recevoir le catalogue sur WhatsApp
            </button>

            <p className="text-center text-xs text-gray-700">
              {COPY.leadReassurance} En continuant, vous acceptez d&apos;être contacté sur WhatsApp au sujet de ce catalogue.
            </p>
          </form>
        )}
      </div>
    </dialog>
  );
}

import { ImageOff } from "lucide-react";

import { useState } from "react";

/** next/image avec repli : si la photo est absente ou cassée, on affiche un emplacement neutre au lieu d'une image brisée. */
export function SafeImage(props: ImageProps) {
  const [failed, setFailed] = useState(false);

  if (failed) {
    return (
      <div
        role={props.alt ? "img" : undefined}
        aria-label={props.alt || undefined}
        aria-hidden={props.alt ? undefined : true}
        className="absolute inset-0 flex items-center justify-center bg-emerald-50 text-emerald-800"
      >
        <ImageOff aria-hidden className="size-6" />
      </div>
    );
  }
  return <Image {...props} onError={() => setFailed(true)} />;
}

import { CATEGORY_SLUGS, HEALTH_GOAL_SLUGS } from "./config";
import { CATEGORIES, HEALTH_GOALS, type Category, type HealthGoal, type Product } from "./types";

export interface CatalogueQuery {
  category?: Category;
  goal?: HealthGoal;
  q?: string;
}

type RawParams = Record<string, string | string[] | undefined>;

const first = (value: string | string[] | undefined) => (Array.isArray(value) ? value[0] : value);
const fold = (text: string) => text.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase();

/** Lit ?categorie=…&besoin=…&q=… ; les valeurs inconnues sont ignorées. */
export function parseCatalogueQuery(params: RawParams): CatalogueQuery {
  const q = first(params.q)?.trim().slice(0, 80);
  return {
    category: CATEGORIES.find((category) => CATEGORY_SLUGS[category] === first(params.categorie)),
    goal: HEALTH_GOALS.find((goal) => HEALTH_GOAL_SLUGS[goal] === first(params.besoin)),
    q: q || undefined,
  };
}

/** Filtre et trie dans l'ordre officiel de la grille tarifaire. */
export function filterProducts(products: readonly Product[], query: CatalogueQuery): Product[] {
  const needle = query.q ? fold(query.q) : null;
  return products
    .filter((product) => !query.category || product.category === query.category)
    .filter((product) => !query.goal || product.healthGoals.includes(query.goal))
    .filter(
      (product) =>
        !needle || fold(`${product.name} ${product.shortDesc} ${product.category}`).includes(needle),
    )
    .sort((a, b) => CATEGORIES.indexOf(a.category) - CATEGORIES.indexOf(b.category));
}

export function buildCatalogueHref(query: CatalogueQuery): string {
  const params = new URLSearchParams();
  if (query.category) params.set("categorie", CATEGORY_SLUGS[query.category]);
  if (query.goal) params.set("besoin", HEALTH_GOAL_SLUGS[query.goal]);
  if (query.q) params.set("q", query.q);
  const search = params.toString();
  return search ? `/catalogue?${search}` : "/catalogue";
}

import { HEALTH_GOAL_SLUGS } from "./config";
import { HEALTH_GOALS, type HealthGoal } from "./types";

/** D'où vient le signal de besoin : plus il est explicite, plus il pèse. */
export type SignalOrigin = "choix" | "pub" | "catalogue" | "panier" | "produit";
export type GoalScores = Partial<Record<HealthGoal, number>>;
export type GoalOrigins = Partial<Record<HealthGoal, SignalOrigin[]>>;

export const SIGNAL_WEIGHTS: Record<SignalOrigin, number> = {
  choix: 100,
  pub: 5,
  catalogue: 3,
  panier: 2,
  produit: 1,
};

export const ORIGIN_LABELS: Record<SignalOrigin, string> = {
  choix: "choisi dans le pop-up",
  pub: "publicité",
  catalogue: "filtre catalogue",
  panier: "panier",
  produit: "fiche produit",
};

const KEYWORDS: ReadonlyArray<readonly [HealthGoal, RegExp]> = [
  ["Digestion & Ventre plat", /digest|ventre|intestin|transit/],
  ["Articulations & Mobilité", /articul|mobilit|\bdos\b|genou/],
  ["Vitalité & Forme Masculine", /vitalit|masculin|homme|energie|énergie/],
  ["Peau & Teint Éclatant", /peau|teint|beaut/],
  ["Détox & Perte de Poids", /detox|détox|minceur|poids|\bc9\b|clean ?9/],
  ["Défenses Immunitaires", /immun|defense|défense|tonus/],
];

/** Reconnaît un besoin dans un slug (?besoin=digestion) ou un nom de campagne (utm_campaign=digestion-30-jours). */
export function goalFromText(text: string | null | undefined): HealthGoal | null {
  if (!text) return null;
  const value = text.toLowerCase();
  const bySlug = HEALTH_GOALS.find((goal) => HEALTH_GOAL_SLUGS[goal] === value);
  if (bySlug) return bySlug;
  return KEYWORDS.find(([, pattern]) => pattern.test(value))?.[0] ?? null;
}

export function applySignal(
  scores: GoalScores,
  origins: GoalOrigins,
  goal: HealthGoal,
  origin: SignalOrigin,
): { scores: GoalScores; origins: GoalOrigins } {
  const known = origins[goal] ?? [];
  return {
    scores: { ...scores, [goal]: (scores[goal] ?? 0) + SIGNAL_WEIGHTS[origin] },
    origins: { ...origins, [goal]: known.includes(origin) ? known : [...known, origin] },
  };
}

/** Besoins classés du plus au moins probable (à égalité : ordre officiel). */
export function rankGoals(scores: GoalScores): HealthGoal[] {
  return HEALTH_GOALS.filter((goal) => (scores[goal] ?? 0) > 0).sort(
    (a, b) => (scores[b] ?? 0) - (scores[a] ?? 0),
  );
}

export interface NeedProfile {
  need: HealthGoal | null;
  origins: SignalOrigin[];
  others: HealthGoal[];
}

/** Synthèse au moment du lead : parcours du visiteur + panier + choix explicite éventuel. */
export function buildNeedProfile(input: {
  scores: GoalScores;
  origins: GoalOrigins;
  chosen: HealthGoal | null;
  cartGoals: readonly HealthGoal[];
}): NeedProfile {
  let state = { scores: input.scores, origins: input.origins };
  for (const goal of input.cartGoals) state = applySignal(state.scores, state.origins, goal, "panier");
  if (input.chosen) state = applySignal(state.scores, state.origins, input.chosen, "choix");

  const [need = null, ...others] = rankGoals(state.scores);
  return { need, origins: need ? (state.origins[need] ?? []) : [], others: others.slice(0, 3) };
}

import { LEAD_DISMISS_DAYS } from "./config";
import { parseLeadPayload } from "./leads";
import type { LeadPayload } from "./types";

const PENDING_KEY = "forever-lead-pending-v1";
const STATUS_KEY = "forever-lead-status-v1";
type LeadStatus = "submitted" | "dismissed";

function safeGet(key: string): string | null {
  try {
    return localStorage.getItem(key);
  } catch {
    return null;
  }
}

function safeSet(key: string, value: string): void {
  try {
    localStorage.setItem(key, value);
  } catch {
    /* navigation privée : on continue sans mémoire */
  }
}

/** « submitted » pour toujours, « dismissed » pendant LEAD_DISMISS_DAYS jours : ensuite le pop-up peut revenir une fois. */
export function readLeadStatus(): LeadStatus | null {
  const raw = safeGet(STATUS_KEY);
  if (!raw) return null;
  try {
    const { status, at } = JSON.parse(raw) as { status?: unknown; at?: unknown };
    if (status === "submitted") return "submitted";
    if (status === "dismissed" && typeof at === "number") {
      return Date.now() - at < LEAD_DISMISS_DAYS * 86_400_000 ? "dismissed" : null;
    }
  } catch {
    /* valeur corrompue : ignorée */
  }
  return null;
}

export function writeLeadStatus(status: LeadStatus): void {
  if (readLeadStatus() === "submitted") return;
  safeSet(STATUS_KEY, JSON.stringify({ status, at: Date.now() }));
}

async function post(payload: LeadPayload, honeypot = ""): Promise<void> {
  try {
    const response = await fetch("/api/leads", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ ...payload, website: honeypot }),
      keepalive: true, // la requête aboutit même si WhatsApp prend le dessus sur l'onglet
    });
    if (response.ok) localStorage.removeItem(PENDING_KEY);
  } catch {
    /* réseau coupé : le lead reste en attente et repart à la prochaine visite */
  }
}

/** Envoie le lead sans jamais bloquer le visiteur ; garde une copie locale jusqu'à confirmation. */
export function sendLead(payload: LeadPayload, honeypot = ""): void {
  safeSet(PENDING_KEY, JSON.stringify(payload));
  void post(payload, honeypot);
}

export function retryPendingLead(): void {
  const raw = safeGet(PENDING_KEY);
  if (!raw) return;
  try {
    const payload = parseLeadPayload(JSON.parse(raw));
    if (payload) void post(payload);
  } catch {
    /* ignoré */
  }
}

export function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

const text = (value: unknown, max: number): string =>
  typeof value === "string" ? value.trim().slice(0, max) : "";

const isGoal = (value: unknown): value is HealthGoal =>
  typeof value === "string" && (HEALTH_GOALS as readonly string[]).includes(value);

/** Validation côté serveur : tout ce qui vient du navigateur est non fiable. */
export function parseLeadPayload(input: unknown): LeadPayload | null {
  if (!isRecord(input)) return null;
  const firstName = text(input.firstName, 60);
  const phone = typeof input.phone === "string" ? normalizeCongoPhone(input.phone) : null;
  if (firstName.length < 2 || !phone) return null;

  return {
    firstName,
    phone,
    need: isGoal(input.need) ? input.need : null,
    needOrigin: text(input.needOrigin, 80),
    otherNeeds: Array.isArray(input.otherNeeds) ? input.otherNeeds.filter(isGoal).slice(0, 5) : [],
    viewedProducts: Array.isArray(input.viewedProducts)
      ? input.viewedProducts.map((name) => text(name, 80)).filter(Boolean).slice(0, 8)
      : [],
    cartTotal:
      typeof input.cartTotal === "number" && Number.isFinite(input.cartTotal)
        ? Math.max(0, Math.round(input.cartTotal))
        : 0,
    campaign: text(input.campaign, 120),
    source: text(input.source, 120),
    referrer: text(input.referrer, 120),
    landingPath: text(input.landingPath, 200),
    pagePath: text(input.pagePath, 200),
  };
}

/**
 * Numéros congolais : 9 chiffres (avec le 0 initial) après l'indicatif 242.
 * Accepte "06 123 4567", "+242 06 123 4567", "00242061234567".
 * Retourne "242061234567" ou null si le numéro est invalide.
 */
export function normalizeCongoPhone(input: string): string | null {
  let digits = input.replace(/\D/g, "");
  if (digits.startsWith("00")) digits = digits.slice(2);
  if (digits.length === 9) digits = `242${digits}`;
  return /^242\d{9}$/.test(digits) ? digits : null;
}

import {
  MAX_LINE_QUANTITY,
  VIP_DISCOUNT_PERCENT,
  VIP_THRESHOLD,
} from "./config";
import type { CartItem, CartTotals, Product } from "./types";

const numberFormatter = new Intl.NumberFormat("fr-FR", {
  maximumFractionDigits: 0,
});

/** 65000 -> "65 000 F" (espaces insécables : le montant ne se coupe jamais en fin de ligne). */
export function formatFCFA(amount: number): string {
  const grouped = numberFormatter.format(Math.round(amount)).replace(/\u202f/g, "\u00a0");
  return `${grouped}\u00a0F`;
}

/** Prix VIP = prix public -5 %, calculé en entiers puis arrondi au franc. */
export function computeVipPrice(pricePublic: number): number {
  return Math.round((pricePublic * (100 - VIP_DISCOUNT_PERCENT)) / 100);
}

export function getUnitPrice(product: Product, vipUnlocked: boolean): number {
  return vipUnlocked ? product.priceVip : product.pricePublic;
}

export function clampQuantity(quantity: number): number {
  if (!Number.isFinite(quantity)) return 1;
  return Math.min(MAX_LINE_QUANTITY, Math.max(1, Math.trunc(quantity)));
}

/**
 * Règle stricte : la remise VIP s'applique à TOUTE la commande si, et seulement si,
 * le total au prix public atteint ou dépasse 65 000 F. En dessous : prix public.
 */
export function computeCartTotals(items: readonly CartItem[]): CartTotals {
  let itemCount = 0;
  let subtotalPublic = 0;
  let subtotalVip = 0;

  for (const { product, quantity } of items) {
    itemCount += quantity;
    subtotalPublic += product.pricePublic * quantity;
    subtotalVip += product.priceVip * quantity;
  }

  const vipUnlocked = subtotalPublic >= VIP_THRESHOLD;
  const total = vipUnlocked ? subtotalVip : subtotalPublic;

  return {
    itemCount,
    subtotalPublic,
    vipUnlocked,
    vipDiscount: subtotalPublic - total,
    total,
    amountToVip: Math.max(0, VIP_THRESHOLD - subtotalPublic),
    progressPercent: Math.min(100, Math.floor((subtotalPublic * 100) / VIP_THRESHOLD)),
  };
}

"use client";

import { useMemo } from "react";
import { create } from "zustand";
import { createJSONStorage, persist } from "zustand/middleware";
import { getProductByCode } from "@/data/products";
import { clampQuantity, computeCartTotals } from "@/lib/pricing";
import type { CartItem, CartLine, CartTotals, Product } from "@/lib/types";

interface CartState {
  lines: CartLine[];
  isOpen: boolean;
  /** Vrai une fois le panier relu depuis localStorage (évite les erreurs d'hydratation). */
  hydrated: boolean;
  addItem: (product: Product, quantity?: number) => void;
  setQuantity: (code: string, quantity: number) => void;
  removeItem: (code: string) => void;
  clear: () => void;
  openCart: () => void;
  closeCart: () => void;
  setHydrated: () => void;
}

interface PersistedCart {
  lines: CartLine[];
}

/** Relecture défensive : ignore les données corrompues et les produits retirés du catalogue. */
function sanitizeLines(value: unknown): CartLine[] {
  if (!Array.isArray(value)) return [];
  const lines: CartLine[] = [];
  for (const entry of value) {
    if (typeof entry !== "object" || entry === null) continue;
    const { code, quantity } = entry as Record<string, unknown>;
    if (typeof code !== "string" || typeof quantity !== "number") continue;
    if (!getProductByCode(code)) continue;
    lines.push({ code, quantity: clampQuantity(quantity) });
  }
  return lines;
}

export const useCartStore = create<CartState>()(
  persist(
    (set) => ({
      lines: [],
      isOpen: false,
      hydrated: false,

      addItem: (product, quantity = 1) =>
        set((state) => {
          const existing = state.lines.find((line) => line.code === product.code);
          const lines = existing
            ? state.lines.map((line) =>
                line.code === product.code
                  ? { ...line, quantity: clampQuantity(line.quantity + quantity) }
                  : line,
              )
            : [...state.lines, { code: product.code, quantity: clampQuantity(quantity) }];
          return { lines, isOpen: true };
        }),

      setQuantity: (code, quantity) =>
        set((state) => ({
          lines:
            quantity <= 0
              ? state.lines.filter((line) => line.code !== code)
              : state.lines.map((line) =>
                  line.code === code ? { ...line, quantity: clampQuantity(quantity) } : line,
                ),
        })),

      removeItem: (code) =>
        set((state) => ({ lines: state.lines.filter((line) => line.code !== code) })),

      clear: () => set({ lines: [] }),
      openCart: () => set({ isOpen: true }),
      closeCart: () => set({ isOpen: false }),
      setHydrated: () => set({ hydrated: true }),
    }),
    {
      name: "forever-cart-v1",
      version: 1,
      storage: createJSONStorage<PersistedCart>(() => localStorage),
      // Hydratation manuelle (voir CartHydrator) : le rendu serveur et le premier rendu client restent identiques.
      skipHydration: true,
      partialize: (state): PersistedCart => ({ lines: state.lines }),
      merge: (persisted, current) => ({
        ...current,
        lines: sanitizeLines((persisted as Partial<PersistedCart> | undefined)?.lines),
      }),
      onRehydrateStorage: () => (state) => {
        state?.setHydrated();
      },
    },
  ),
);

function resolveLines(lines: readonly CartLine[]): CartItem[] {
  return lines.flatMap((line) => {
    const product = getProductByCode(line.code);
    return product ? [{ product, quantity: line.quantity }] : [];
  });
}

/** Lignes du panier avec les prix courants du catalogue. */
export function useCartItems(): CartItem[] {
  const lines = useCartStore((state) => state.lines);
  return useMemo(() => resolveLines(lines), [lines]);
}

/** Totaux avec la règle stricte du palier de 65 000 F. */
export function useCartTotals(): CartTotals {
  const items = useCartItems();
  return useMemo(() => computeCartTotals(items), [items]);
}

export function useCartHydrated(): boolean {
  return useCartStore((state) => state.hydrated);
}

/** Pour les gestionnaires d'événements (ex. confirmer la commande), hors rendu React. */
export function getCartItems(): CartItem[] {
  return resolveLines(useCartStore.getState().lines);
}

/**
 * Mémoire silencieuse du parcours : d'où vient le visiteur et ce qu'il regarde.
 * Rien n'est demandé au visiteur ; ces signaux servent à deviner son besoin au moment du lead.
 */
export const useInterestStore = create<InterestState>()(
  persist(
    (set) => ({
      firstSeenAt: null,
      landingPath: null,
      campaign: null,
      source: null,
      referrer: null,
      lastAdKey: null,
      scores: {},
      origins: {},
      viewedProducts: [],

      captureArrival: ({ search, pathname, referrer }) =>
        set((state) => {
          const params = new URLSearchParams(search);
          const besoin = params.get("besoin");
          const campaign = params.get("utm_campaign");
          const content = params.get("utm_content");
          const source =
            [params.get("utm_source"), params.get("utm_medium")].filter(Boolean).join(" / ") || null;
          const adGoal = goalFromText(besoin) ?? goalFromText(campaign) ?? goalFromText(content);
          const adKey = [besoin, campaign, content].filter(Boolean).join("|") || null;

          const base = {
            firstSeenAt: state.firstSeenAt ?? new Date().toISOString(),
            landingPath: state.landingPath ?? pathname,
            referrer: state.referrer ?? (referrer || null),
            campaign: campaign ?? state.campaign,
            source: source ?? state.source,
          };
          if (adGoal && adKey !== state.lastAdKey) {
            return { ...base, lastAdKey: adKey, ...applySignal(state.scores, state.origins, adGoal, "pub") };

          return base;
        }),

          for (const goal of healthGoals) next = applySignal(next.scores, next.origins, goal, "produit");
          return {
            ...next,
            viewedProducts: [code, ...state.viewedProducts.filter((item) => item !== code)].slice(0, MAX_VIEWED),

        }),
